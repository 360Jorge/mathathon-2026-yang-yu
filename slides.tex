\begin{frame}{The problem}
\begin{itemize}\item A smooth cubic fourfold $X$ has a lattice $A(X)$ of algebraic classes.\item A Hassett divisor $\mathcal C_d$ records a primitive rank-two sublattice containing $h^2$, with discriminant $d$.\item Question: how do we enumerate components of $\mathcal C_{d_1}\cap\mathcal C_{d_2}$?\end{itemize}
\vspace{0.5em}\alert{Scope:} reproduce and explain published results computationally; no claim of a new classification.
\end{frame}
\begin{frame}{Why a rank-three lattice?}
Two rank-two lattices share the class $h^2$:
\[K_8=\begin{pmatrix}3&1\\1&3\end{pmatrix},\qquad K_{14}=\begin{pmatrix}3&1\\1&5\end{pmatrix}.\]
Their span has basis $(h^2,u,v)$ and Gram matrix
\[G_\tau=\begin{pmatrix}3&1&1\\1&3&\tau\\1&\tau&5\end{pmatrix},\qquad \tau=(u,v)\in\mathbb Z.\]
Only one inner product is unknown.
\end{frame}
\begin{frame}{First two filters: positivity and roots}
\[\det G_\tau=37+2\tau-3\tau^2.\]
Sylvester's criterion gives
\[T_1=\{-3,-2,-1,0,1,2,3\}.\]
\begin{itemize}\item At $\tau=3$, $u-v$ has norm $2$.\item At $\tau=-3$, $u+v$ has norm $2$.\item A sum-of-squares bound excludes roots for $|\tau|\le2$.\end{itemize}
\[\boxed{T_2=\{-2,-1,0,1,2\}}\]
\end{frame}
\begin{frame}{The nontrivial part: overlattices}
The span $M_\tau=\langle h^2,u,v\rangle$ may not be saturated.
\begin{enumerate}\item Enumerate isotropic subgroups of $M_\tau^\vee/M_\tau$.\item Construct corresponding integral overlattices.\item Test primitivity of both rank-two labellings, evenness of $(h^2)^\perp$, and absence of roots.\end{enumerate}
\alert{Program output:} for $(8,14)$, two nontrivial subgroups at $\tau=-1$ and three at $\tau=1$; none accepted. Explicit rejection certificates remain a next step.
\end{frame}

\begin{frame}{Connecting the two parameter conventions}
Auel et al. use $T=h^2+v$ rather than $v$.
\[\begin{pmatrix}3&1&1\\1&3&\tau\\1&\tau&5\end{pmatrix}
\quad\longrightarrow\quad
\begin{pmatrix}3&1&4\\1&3&\tau+1\\4&\tau+1&10\end{pmatrix}.\]
Thus $\tau_{\mathrm{ABBVA}}=\tau_{\mathrm{YY}}+1$.
\vspace{0.7em}
\alert{The same five lattices appear in different integral bases.}
\end{frame}
